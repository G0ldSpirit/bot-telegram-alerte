# Toutes les 5 minutes
*/5 * * * * cd /home/user/bot-telegram-alerte && node src/monitor/monitor.js --once

# Toutes les 10 minutes
*/10 * * * * cd /home/user/bot-telegram-alerte && node src/monitor/monitor.js --once

# Toutes les heures
0 * * * * cd /home/user/bot-telegram-alerte && node src/monitor/monitor.js --once

# Tous les jours à 9h00
0 9 * * * cd /home/user/bot-telegram-alerte && node src/monitor/monitor.js --once

# Note: Remplacer /home/user/bot-telegram-alerte par le chemin réel de votre installation
